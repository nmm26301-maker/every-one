cmake_minimum_required(VERSION 3.20)
project(english_local LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_OPENSSL OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
if(LOCAL_ENGINE STREQUAL "whisper")
 add_subdirectory(../vendor/whisper.cpp whisper-vendor)
 add_library(english_whisper SHARED whisper/bridge.cpp)
 target_link_libraries(english_whisper PRIVATE whisper)
 set(LOCAL_TARGET english_whisper)
elseif(LOCAL_ENGINE STREQUAL "speech")
 set(USE_MBROLA OFF CACHE BOOL "" FORCE)
 set(USE_LIBSONIC OFF CACHE BOOL "" FORCE)
 set(USE_LIBPCAUDIO OFF CACHE BOOL "" FORCE)
 set(USE_ASYNC OFF CACHE BOOL "" FORCE)
 set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
 add_subdirectory(../vendor/espeak-ng speech-vendor)
 add_library(english_speech SHARED speech/bridge.cpp)
 add_dependencies(english_speech espeak-ng)
 target_link_libraries(english_speech PRIVATE $<TARGET_FILE:espeak-ng> $<TARGET_FILE:ucd> $<TARGET_FILE:speechPlayer> m)
 target_include_directories(english_speech BEFORE PRIVATE ../vendor/espeak-ng/src/include)
 target_include_directories(english_speech PRIVATE ../vendor/espeak-ng/src/ucd-tools/src/include)
 set(LOCAL_TARGET english_speech)
else()
 add_subdirectory(../vendor/llama.cpp llama-vendor)
 add_library(english_llama SHARED llama/bridge.cpp)
 target_link_libraries(english_llama PRIVATE llama)
 set(LOCAL_TARGET english_llama)
endif()
if(ANDROID)
 target_link_libraries(${LOCAL_TARGET} PRIVATE log android)
 target_link_options(${LOCAL_TARGET} PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic -Wl,-z,max-page-size=16384)
else()
 if(JNI_HEADER_DIR)
  target_include_directories(${LOCAL_TARGET} PRIVATE ${JNI_HEADER_DIR})
 else()
  find_package(JNI REQUIRED COMPONENTS JVM)
  target_include_directories(${LOCAL_TARGET} PRIVATE ${JNI_INCLUDE_DIRS})
 endif()
 target_link_options(${LOCAL_TARGET} PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic)
endif()
